Resource Limits and Requests
# Add this inside spec.template.spec.containers[-]
        resources:
          limits:
            cpu: "500m"
            memory: "256Mi"
          requests:
            cpu: "100m"
            memory: "128Mi"

3. Liveness and Readiness Probes
# Add this inside spec.template.spec.containers[-]
        livenessProbe:
          exec:
            command:
            - python
            - -c
            - "import os; print('Alive')" # Or a custom health check script
          initialDelaySeconds: 15
          periodSeconds: 20
